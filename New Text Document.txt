local function Deny()
    attempts += 1
    status.Text = "Invalid key. Access denied."
    status.TextColor3 = Color3.fromRGB(255, 95, 110)

    local original = panel.Position
    local left = original + UDim2.fromOffset(-7, 0)
    local right = original + UDim2.fromOffset(7, 0)

    task.spawn(function()
        TweenService:Create(panel, TweenInfo.new(0.045), {Position = left}):Play()
        task.wait(0.045)
        TweenService:Create(panel, TweenInfo.new(0.045), {Position = right}):Play()
        task.wait(0.045)
        TweenService:Create(panel, TweenInfo.new(0.045), {Position = original}):Play()
    end)
end

local function Unlock()
    if busy or unlocked then return end
    busy = true

    if input.Text == REQUIRED_KEY then
        unlocked = true
        status.Text = "Access granted. Loading..."
        status.TextColor3 = Color3.fromRGB(52, 211, 153)
        unlock.Text = "ACCESS GRANTED"

        task.wait(0.25)

        -- Remove the temporary Rayfield/key UI after successful unlock.
        -- This keeps the unlocked script clean and prevents the key window
        -- from remaining behind the main menu.
        pcall(function()
            local containers = {}

            local okHui, hui = pcall(function()
                return gethui and gethui()
            end)
            if okHui and hui then
                table.insert(containers, hui)
            end

            local okCore, coreGui = pcall(function()
                return game:GetService("CoreGui")
            end)
            if okCore and coreGui then
                table.insert(containers, coreGui)
            end

            local okPlayerGui, playerGui = pcall(function()
                return LocalPlayer:FindFirstChildOfClass("PlayerGui")
            end)
            if okPlayerGui and playerGui then
                table.insert(containers, playerGui)
            end

            for _, container in ipairs(containers) do
                for _, gui in ipairs(container:GetChildren()) do
                    local guiName = string.lower(tostring(gui.Name))
                    if guiName == "rayfield"
                        or guiName:find("rayfield", 1, true)
                        or guiName == "plesnevi yxurayfield"
                        or guiName == "plesnevi yxukeygate" then
                        pcall(function()
                            gui:Destroy()
                        end)
                    end
                end
            end
        end)

        keyGui:Destroy()
    else
        Deny()
        busy = false
    end
end

-- ==================== MAX GREEN FORCING ====================
local function forcePerfectShot()
    if not enabled then return end

    local pgui = player.PlayerGui
    local visual = pgui:FindFirstChild("Visual") or pgui:FindFirstChildWhichIsA("ScreenGui")
    if visual then
        local shooting = visual:FindFirstChild("Shooting")
        if shooting then
            local bar = shooting:FindFirstChild("Bar") or shooting:FindFirstChild("Meter")
            if bar and bar:IsA("Frame") then
                bar.Size = UDim2.fromScale(1, 1)
                TweenService:Create(bar, TweenInfo.new(0.005), {Size = UDim2.fromScale(1, 1)}):Play()
            end
        end
    end

    pcall(function()
        local char = player.Character
        if char then
            for _, obj in pairs(char:GetDescendants()) do
                local n = obj.Name:lower()
                if n:find("contest") or n:find("smother") or n:find("shot") or n:find("meter") 
                   or n:find("release") or n:find("range") or n:find("quality") 
                   or n:find("perfect") or n:find("great") or n:find("late") or n:find("early") then
                    
                    if obj:IsA("NumberValue") or obj:IsA("IntValue") or obj:IsA("FloatValue") then
                        obj.Value = perfectValue
                    elseif obj:IsA("BindableEvent") then
                        obj:Fire(perfectValue)
                    end
                end
            end
        end
    end)
end

-- ==================== AUTO GUARD ====================
local guardConnection
local function startAutoGuard()
    if guardConnection then guardConnection:Disconnect() end
    guardConnection = RunService.Heartbeat:Connect(function()
        if not autoGuardEnabled then return end
        local myChar = player.Character
        if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return end
        local myRoot = myChar.HumanoidRootPart
        local closest, minDist = nil, math.huge

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= player and plr.Character then
                local root = plr.Character:FindFirstChild("HumanoidRootPart")
                if root then
                    local hasBall = plr.Character:FindFirstChild("Basketball") or plr.Character:FindFirstChildWhichIsA("Tool")
                    if hasBall then
                        local dist = (myRoot.Position - root.Position).Magnitude
                        if dist < minDist and dist < 60 then
                            minDist = dist
                            closest = root
                        end
                    end
                end
            end
        end

        if closest then
            local dir = (closest.Position - myRoot.Position).Unit
            myRoot.CFrame = CFrame.lookAt(myRoot.Position, myRoot.Position + dir)
        end
    end)
end
